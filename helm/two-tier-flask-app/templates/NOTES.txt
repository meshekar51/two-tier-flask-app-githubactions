Open the app:
{{- if .Values.flask.ingress.enabled }}
{{- range .Values.flask.ingress.hosts }}
  http{{ if $.Values.flask.ingress.tls }}s{{ end }}://{{ .host }}
{{- end }}
{{- else if eq .Values.flask.service.type "NodePort" }}
  NODE_PORT=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "twotier.fullname" . }} -o jsonpath='{.spec.ports[0].nodePort}')
  NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="InternalIP")].address}')
  echo "http://$NODE_IP:$NODE_PORT"
{{- else if eq .Values.flask.service.type "LoadBalancer" }}
  The load balancer can take a few minutes to get an address:
  kubectl -n {{ .Release.Namespace }} get svc {{ include "twotier.fullname" . }} -w
  echo "http://$(kubectl -n {{ .Release.Namespace }} get svc {{ include "twotier.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].hostname}{.status.loadBalancer.ingress[0].ip}')"
{{- else }}
  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "twotier.fullname" . }} 8080:{{ .Values.flask.service.port }}
  then browse http://localhost:8080
{{- end }}
